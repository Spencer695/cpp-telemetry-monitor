cmake_minimum_required(VERSION 3.16)
project(CppTelemetryMonitor VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(telemetry_core src/protocol.cpp)
target_include_directories(telemetry_core PUBLIC include)
target_link_libraries(telemetry_core PUBLIC Threads::Threads)
add_library(telemetry_server src/server.cpp)
target_link_libraries(telemetry_server PUBLIC telemetry_core)
add_executable(telemetry_monitor src/main.cpp)
target_link_libraries(telemetry_monitor PRIVATE telemetry_server)
if(WIN32)
  target_link_libraries(telemetry_server PUBLIC ws2_32)
endif()
set(TELEMETRY_SANITIZER "none" CACHE STRING "none, address, or thread")
set_property(CACHE TELEMETRY_SANITIZER PROPERTY STRINGS none address thread)
if(NOT TELEMETRY_SANITIZER MATCHES "^(none|address|thread)$")
  message(FATAL_ERROR "Unknown TELEMETRY_SANITIZER")
endif()
add_library(telemetry_options INTERFACE)
if(MSVC)
  target_compile_options(telemetry_options INTERFACE /W4 /WX /permissive- /utf-8 /Zc:__cplusplus)
  if(NOT TELEMETRY_SANITIZER STREQUAL "none")
    message(FATAL_ERROR "Sanitizer configurations in this project require GCC or Clang on Linux")
  endif()
else()
  target_compile_options(telemetry_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
  if(TELEMETRY_SANITIZER STREQUAL "address")
    target_compile_options(telemetry_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
    target_link_options(telemetry_options INTERFACE -fsanitize=address,undefined -no-pie)
  elseif(TELEMETRY_SANITIZER STREQUAL "thread")
    target_compile_options(telemetry_options INTERFACE -fsanitize=thread -fno-omit-frame-pointer -fno-pie)
    target_link_options(telemetry_options INTERFACE -fsanitize=thread -no-pie)
  endif()
endif()
target_link_libraries(telemetry_core PUBLIC telemetry_options)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE telemetry_core)
  add_test(NAME core COMMAND core_tests)
  set_tests_properties(core PROPERTIES TIMEOUT 30)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME tcp_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:telemetry_monitor>")
  set_tests_properties(tcp_integration PROPERTIES TIMEOUT 90)
endif()
option(TELEMETRY_BUILD_GUI "Build the optional Qt 6 Widgets dashboard" OFF)
if(TELEMETRY_BUILD_GUI)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Network)
  add_library(telemetry_dashboard gui/dashboard.cpp gui/dashboard.hpp)
  set_target_properties(telemetry_dashboard PROPERTIES AUTOMOC ON)
  target_include_directories(telemetry_dashboard PUBLIC gui)
  target_link_libraries(telemetry_dashboard PUBLIC telemetry_server Qt6::Widgets Qt6::Network)
  add_executable(telemetry_gui gui/main.cpp)
  target_link_libraries(telemetry_gui PRIVATE telemetry_dashboard)
  if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test Network)
    add_executable(gui_tests tests/gui_tests.cpp)
    set_target_properties(gui_tests PROPERTIES AUTOMOC ON)
    target_link_libraries(gui_tests PRIVATE telemetry_dashboard Qt6::Test Qt6::Network)
    add_test(NAME qt_dashboard COMMAND gui_tests)
    set_tests_properties(qt_dashboard PROPERTIES TIMEOUT 45 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
endif()
